Python: Create Post

import requests

token = "YOUR_TOKEN_HERE"
url = "https://rgavanp.kdns.fr/blog/api/posts/"
payload = {
    "title": "API Post",
    "slug": "api-post-example-change-this-slug",
    "body": "posted from python api",
    "status": "PB",
    "tags": ["#api", "#python"],
}
response = requests.post(url, headers={"Authorization": f"Token {token}", "Content-Type": "application/json"}, json=payload, timeout=30)
print(response.status_code)
print(response.text)

Python: Edit Post

import requests

token = "YOUR_TOKEN_HERE"
post_id = 1
url = f"https://rgavanp.kdns.fr/blog/api/posts/{post_id}/"
payload = {"title": "Updated title from python", "body": "updated body from python api"}
response = requests.patch(url, headers={"Authorization": f"Token {token}", "Content-Type": "application/json"}, json=payload, timeout=30)
print(response.status_code)
print(response.text)

Python: Delete Comment

import requests

token = "YOUR_TOKEN_HERE"
headers = {"Authorization": f"Token {token}"}
comment_response = requests.delete("https://rgavanp.kdns.fr/blog/api/comments/1/", headers=headers, timeout=30)
print("comment", comment_response.status_code, comment_response.text)
